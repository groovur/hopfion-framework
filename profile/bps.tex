\begin{proposition}[BPS ratio]
\label{P1:prop:bps}
For the standard BPS profile $f=2\arctan(a/\rho)$, the thin-torus ratio
$I_4/I_{2a}$ is scale-invariant (independent of $a$) and equals
\begin{equation}
  \left.\frac{I_4}{I_{2a}}\right|_{\mathrm{BPS}}
  = \frac{32\,\Gamma(2)\Gamma(4)/\Gamma(6)}
         {64\,\Gamma(3)^2/\Gamma(6)}
  = \frac{3}{4}.
  \label{P1:eq:bps_ratio}
\end{equation}
\end{proposition}
\status{published}
\source{P1:prop:bps}
\residual{exact}

\begin{definition}[Modified BPS]
\label{P1:def:modified_bps}
For $C>1/3$, the modified BPS profile
$f = 2\arctan\!\bigl((a/\rho)^C\bigr)$
satisfies $f' = -C\sin f/\rho$ exactly.
\end{definition}
\status{published}
\source{P1:def:modified_bps}

\begin{theorem}[Modified BPS ratio]
\label{P1:thm:modBPS}
For the modified BPS profile $f = 2\arctan\!\bigl((a/\rho)^C\bigr)$
with exponent $C > 1/3$,
\begin{equation}
  \boxed{\frac{I_4}{I_{2a}}(C)
  = \frac{C(2C+1)}{(C^2+1)(3C-1)}.}
  \label{P1:eq:modBPS}
\end{equation}
This is scale-invariant (independent of $a$), and at $C=1$ gives $3/4$.
\end{theorem}
\status{published}
\source{P1:thm:modBPS}
\begin{proof}
With $u=(a/\rho)^C$ and the substitution $\rho=a/u^{1/C}$,
\begin{align}
  I_4   &= 64C\,a^{2/C-2}\cdot\tfrac{1}{2}\,B\!\left(3+\tfrac{1}{C},\;3-\tfrac{1}{C}\right), \\
  I_{2a} &= 64(C^2+1)\,a^{2/C-2}\cdot\tfrac{1}{2}\,B\!\left(2+\tfrac{1}{C},\;4-\tfrac{1}{C}\right).
\end{align}
The common factor $a^{2/C-2}$ cancels.  Using $\Gamma(n+1)=n\Gamma(n)$:
\[
  \frac{B(3+1/C,\;3-1/C)}{B(2+1/C,\;4-1/C)}
  = \frac{\Gamma(3+1/C)\Gamma(3-1/C)}{\Gamma(2+1/C)\Gamma(4-1/C)}
  = \frac{(2+1/C)}{(3-1/C)}.
\]
Therefore $I_4/I_{2a} = C(2+1/C)/((C^2+1)(3-1/C)) = C(2C+1)/((C^2+1)(3C-1))$.
\end{proof}
\status{published}
\source{P1:thm:modBPS}
\residual{exact formula; at $C=C^*\approx3.432$ gives $2^{4/3}/\vp^5$ to $0.020\%$}

\begin{proposition}[Amplitude Pohozaev integrals]
\label{P3:prop:pohozaev_a}
For any smooth $Q=2$ profile $f$, the Pohozaev variation $\xi=\sin f$ gives:
\begin{align}
  \int \sin f\cdot\frac{\delta J_{2a}}{\delta f}\,dV &= 6J_{2a}^{cf}, \label{P3:eq:poh_J2a}\\
  \int \sin f\cdot\frac{\delta \Jfb}{\delta f}\,dV &= 2J_{\mathrm{fb}}^{cf2}, \label{P3:eq:poh_Jfb}\\
  \int \sin f\cdot\frac{\delta J_4}{\delta f}\,dV &= 6J_4^{cf}, \label{P3:eq:poh_J4}
\end{align}
where $J_{2a}^{cf} = \int\cos f\cdot\sin^4\!f\cdot\kk\,dV$, etc.
\end{proposition}
\status{published}
\source{P3:prop:pohozaev_a}

\begin{remark}
\label{P3:rem:factor6_vs_2}
The factor 6 in~\eqref{P3:eq:poh_J2a} and~\eqref{P3:eq:poh_J4} vs.\ factor 2
in~\eqref{P3:eq:poh_Jfb} reflects the different feedback structure:
$J_{2a}$ and $J_4$ have $\sin^4\!f$ or $\sin^2\!f$ prefactors (both
giving IBP contributions of $+2$), while $J_{\mathrm{fb}}$ has the
nonlinear feedback weight $f_\beta^2$ which does not contribute an
additional factor.
\end{remark}
\status{published}
\source{P3:rem:factor6_vs_2}

At the EL saddle of $E_{\mathrm{fb}} = \Kfb\cdot\Jfour$, the Pohozaev

which yields
\begin{equation}
  V^* \;=\; \frac{-3(a+b)}{\mu^*\,(c+3b)}.
  \label{P3:eq:V_from_poh_A}
\end{equation}

\begin{proposition}[Main Pohozaev identity]
\label{P3:prop:star_identity}
$V^*=\vp$ if and only if at the EL saddle:
\begin{equation}
  \sqrt{k+2}\,\frac{J_{\mathrm{fb}}^{cf2}}{\Jfb}
  + 3\!\left(\frac{J_{2a}^{cf}}{\Ja} + 2\vp\,\frac{J_4^{cf}}{\Jfour}\right) = 0.
  \label{P3:eq:star_identity}
\end{equation}
\end{proposition}
\status{proved (identity exact; $\vp$ selection requires WZW input from Theorem~\ref{P3:thm:bos_main})}
\source{P3:prop:star_identity}

\begin{remark}[Physical interpretation]
\label{P3:rem:star_identity_interp}
Identity~\eqref{P3:eq:star_identity} relates the \emph{cos-weighted} averages of
the three curvature functionals.  The cos-weight $\cos f$ is the angular factor
in the spin-connection of the Hopfion field $n: \mathbb{R}^3\to S^2$, and the
identity is a constraint on how the profile distributes curvature between
tube, winding, and feedback channels at the WZW-selected $\mu^*$.
\end{remark}
\status{published}
\source{P3:rem:star_identity_interp}

\paragraph{Variation (d): Radial dilation, and the $\vp^9\sqrt{\Ja\Jfour}=Q$ conjecture.}
The Pohozaev virial identity for the tube-expansion generator
$\xi_b=\chi=(r-R_0)f_r+zf_z$ is
\begin{equation}
  J_{2a}\,N_a + \mu^*\,J_{\mathrm{fb}}\,N_{\mathrm{fb}} + K_{\mathrm{fb}}\,N_4 = 0,
  \label{P3:eq:pohozaev_b}
\end{equation}
where $N_a,N_4,N_{\mathrm{fb}}$ are the tube-scaling exponents, rearranging to
\begin{equation}
  V^* = -\frac{N_a + N_4}{\mu^*\,(N_{\mathrm{fb}} + N_4)}.
  \label{P3:eq:virial_V}
\end{equation}
For the radial-dilation generator $\xi_d = r\,\partial_r f$, substituting the
Derrick balance and $P_{\Ja}=-\Ja$ into $\Jfour P_{\Kfb}+\Kfb P_{\Jfour}=0$ gives
\begin{equation}
  \Ja = \frac{\vp^5\,\mathcal{B}(R_0)}{2\,(1+2^{4/3})},
  \label{P3:eq:Ja_from_poh_d}
\end{equation}
where $\mathcal{B}(R_0)=\oint_{\partial B_\varepsilon}F_{13}\cdot(r-R_0)\rho^{-1}dS$
is the Hopf flux of $F_{13}$ through the core tube.  The conjecture

  \label{P3:eq:B_required}
\end{equation}
